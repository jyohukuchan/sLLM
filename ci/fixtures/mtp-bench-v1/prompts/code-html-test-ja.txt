    <script type="module">
      const percent = (value) =>
        value === null ? "—" : `${(value * 100).toFixed(2)}%`;
      const points = (value) =>
        value === null ? "—" : `${value >= 0 ? "+" : ""}${(value * 100).toFixed(2)} pt`;

      function renderRow(row) {
        const tr = document.createElement("tr");
        const delta = document.createElement("td");
        delta.textContent = points(row.acceptance_delta);
        delta.className = row.acceptance_delta >= 0 ? "delta-positive" : "delta-negative";
        for (const text of [row.cond_id, row.code_language ?? "—", row.blocks,
                            `${row.accepted} / ${row.proposed}`, percent(row.acceptance)]) {
          const td = document.createElement("td");
          td.textContent = text;
          tr.append(td);
        }
        tr.append(delta);
        const tps = document.createElement("td");
        tps.textContent = row.decode_tokens_per_second.toFixed(3);
        tr.append(tps);
        return tr;
      }

      async function load() {
        const response = await fetch("./results.json", { cache: "no-store" });
        if (!response.ok) {
          throw new Error(`results.json returned ${response.status}`);
        }
        const data = await response.json();
        document.getElementById("campaign-id").textContent = data.campaign_id;
        document.getElementById("exact-target").textContent = data.exact_target;
        document.getElementById("tile-acceptance").textContent = percent(data.tier_a.acceptance);
        document.getElementById("tile-decode").textContent =
          data.tier_a.decode_tokens_per_second.toFixed(3);
        document.getElementById("tile-draft").textContent =
          data.tier_a.draft_ms_per_block.toFixed(3);

        const drift = document.getElementById("tile-drift");
        drift.textContent = points(data.baseline_drift);
        drift.classList.toggle("warn", Math.abs(data.baseline_drift) > 0.015);

        const tierA = document.querySelector("#tier-a tbody");
        for (const row of data.tier_a.rows) {
          tierA.append(renderRow(row));
        }
        const tierB = document.querySelector("#tier-b tbody");
        for (const row of data.tier_b.rows) {
          const tr = document.createElement("tr");
          for (const text of [row.cond_id, percent(row.acceptance),
                              row.decode_tokens_per_second.toFixed(3)]) {
            const td = document.createElement("td");
            td.textContent = text;
            tr.append(td);
          }
          tierB.append(tr);
        }
      }

      load().catch((error) => {
        document.querySelector(".subtitle").textContent = `Load failed: ${error.message}`;
      });
    </script>
